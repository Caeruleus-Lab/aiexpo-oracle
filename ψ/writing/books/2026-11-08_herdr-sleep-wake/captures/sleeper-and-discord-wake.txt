  python3 skills/team-wake/sleeper.py list                      # asleep cards on this machine
  python3 skills/team-wake/sleeper.py list --all                # this machine + every machine in oracles/*/config.yaml
  python3 skills/team-wake/sleeper.py wake duang-oracle ...     # wake (over ssh when the card's machine is another one)
  just sleepers  ·  just wake-sleeper duang-oracle

Cards are herdr-sleep's (~/.local/state/herdr-sleep/<day>/<label>.json). herdr-sleep's own `wake` always opens a NEW
space at the bottom of the sidebar and runs the card's bare argv, which has no bot token and no Claude account. This
wakes the way the fleet launches: in the KEPT space when it is still there with no agent in it (else a fresh one, a
worktree nested under its team), with wake.launch() for a bot owner (bot token + claude_token from pass on the launch
line, --channels) or the plain account launch for the rest, then renames the agent and hands it its own bookmark.
"""

def wake_one(label):
    got = cards().get(label)
    if not got:
        return f"{label}: no card on this machine"
    path, c = got
    if not asleep(c):
        return f"{label}: not asleep (woke {c.get('woke')})"
    cwd, sid = c["space"]["cwd"], c["agent"]["session"]
    if w.claude_in(cwd):
        return f"{label}: a Claude already runs in {cwd}; not starting a second body"
    pane = kept_pane(c["space"]["id"], cwd) or open_space(label, cwd)
    if not pane:
        return f"{label}: could not find or open a space at {cwd}"
    cfg = w.configs().get(label, {})
    if c["agent"].get("channel") is not False and w.has_token(label) and "--channels" in " ".join(c["agent"]["argv"]):
        cmd = w.launch(label, cfg, cwd, sid)                       # bot owner: token + account on the launch line
    else:
        tok = cfg.get("claude_token") or c["agent"].get("token") or "dd2"
        cmd = (f"cd '{cwd}' && env -u DISCORD_BOT_TOKEN -u DISCORD_STATE_DIR "
               f"CLAUDE_CODE_OAUTH_TOKEN=\"$(pass show claude/token-{tok} | head -1)\" CLAUDE_TOKEN_NAME={tok} "
               f"claude --dangerously-skip-permissions --resume {sid}")
    w.herdr("pane", "run", pane, cmd)
    w.sh("herdr", "--session", w.SESSION, "pane", "wait-output", pane, "--match", "permissions", "--timeout", "90000")
    time.sleep(6)
    w.herdr("agent", "rename", pane, c["agent"].get("name") or label)
    bm = (c.get("bookmark") or "").strip()
    if bm:
        w.herdr("agent", "prompt", pane, "[sleeper] Woke from sleep. Your own bookmark from before: " + " ".join(bm.split()))
    c["woke"] = datetime.datetime.now().astimezone().isoformat(timespec="minutes")
    c["woke_in"] = pane
    json.dump(c, open(path, "w"), indent=2, ensure_ascii=False)
    bot = w.discord_alive(cwd) if "--channels" in cmd else None
    return f"{label}: awake in {pane}" + ("" if bot is None else f", bot {'on' if bot else 'NOT on'}")
